# lists what the inference path really imports. torch / torchaudio are taken
# from the environment (Kaggle / Colab stock builds).
#
# Instead of the second command you can clone the repository and point the
# provider at it:  export ABM_INDEXTTS_REPO=/path/to/index-tts
#

# imports under transformers 4.52.x (it fails at import under 4.57.x). It

# separate environment or notebook session for each engine. transformers

# HuggingFace on first use (no token, not gated): about 4.3 GB for
